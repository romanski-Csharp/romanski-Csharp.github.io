How I write tests

public class LoginPage(IPage page)
{
    private ILocator Email    => page.GetByLabel("Email");
    private ILocator Password => page.GetByLabel("Password");
    private ILocator SignIn   => page.GetByRole(AriaRole.Button, new() { Name = "Sign in" });
    public  ILocator Error    => page.GetByTestId("login-error");

    public async Task OpenAsync() => await page.GotoAsync("/login");

    public async Task LoginAsync(string email, string password)
    {
        await Email.FillAsync(email);
        await Password.FillAsync(password);
        await SignIn.ClickAsync();
    }
}
[TestFixture, Category("Regression")]
public class LoginTests : PageTest
{
    private LoginPage _login = null!;

    [SetUp]
    public async Task SetUp()
    {
        _login = new LoginPage(Page);
        await _login.OpenAsync();
    }

    [Test]
    public async Task Login_WithValidCredentials_RedirectsToLobby()
    {
        await _login.LoginAsync(TestUsers.Player.Email, TestUsers.Player.Password);
        await Expect(Page).ToHaveURLAsync(new Regex("/lobby"));
    }

    [TestCase("wrong@mail.com", "badpass")]
    public async Task Login_WithInvalidCredentials_ShowsError(string email, string pass)
    {
        await _login.LoginAsync(email, pass);
        await Expect(_login.Error).ToBeVisibleAsync();
    }
}

[TestFixture, Category("API")]
public class GamesApiTests : PlaywrightTest
{
    private IAPIRequestContext _api = null!;

    [SetUp]
    public async Task SetUp() =>
        _api = await Playwright.APIRequest.NewContextAsync(new() { BaseURL = Config.ApiUrl });

    [Test]
    public async Task GetGames_ReturnsOkAndNonEmptyList()
    {
        var response = await _api.GetAsync("/api/games");

        Assert.That(response.Status, Is.EqualTo(200));
        var games = await response.JsonAsync<List<GameDto>>();
        Assert.That(games, Is.Not.Empty);
        Assert.That(games, Has.All.Matches<GameDto>(g => !string.IsNullOrEmpty(g.Name)));
    }

    [TearDown]
    public async Task TearDown() => await _api.DisposeAsync();
}
